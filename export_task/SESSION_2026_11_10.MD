from __future__ import annotations

import struct
from typing import Optional


# =============================================================================
# Confluent Wire Format stripping
# =============================================================================

def strip_confluent_header(raw: bytes) -> bytes:
    """
    Strip 5-byte Confluent Schema Registry header nếu có magic byte 0x00.

    Format: [0x00][schema_id: 4 bytes big-endian][payload]
    Nếu không có magic byte → dữ liệu là plain JSON → trả nguyên.
    """
    if len(raw) > 5 and raw[0] == 0x00:
        return raw[5:]
    return raw


# =============================================================================
# Input model — raw Kafka trade event
# =============================================================================

class TradeEvent:
    """
    Immutable value object cho một trade event từ Kafka.

    Không dùng Pydantic ở đây vì class này được serialize vào Spark UDF —

        self,
        symbol: str,
        price: float,
        size: float,
        side: str,
        ts_ms: int,
    ) -> None:

        self.price    = price
        self.size     = size
        self.side     = side
        self.ts_ms    = ts_ms

        Đây là CANONICAL BOUNDARY duy nhất xử lý bytes từ Kafka.
        Mọi deserialization logic nằm ở đây, không rải trong job code.

def _write_to_influxdb(df_pandas, batch_id: int) -> None:
    """
    foreachBatch sink: ghi VWAP rows vào InfluxDB bằng influxdb-client.

    df_pandas columns: symbol, window_start, window_end, interval,
                       vwap, total_volume, num_trades

    InfluxDB point schema (phải khớp với GetVWAP Flux query trong backend):
        measurement: "vwap"
        tag:   symbol, interval
        field: vwap (float), volume (float), num_trades (integer)
        time:  window_end (UTC nanoseconds)

    Lý do dùng raw line protocol thay vì Point builder:
        Line protocol string concat nhanh hơn ~3x trên hot path (L23).
    """
    if df_pandas.empty:
        return

    from dataops_processing.config.settings import get_settings
    cfg = get_settings().influxdb

    try:
        from influxdb_client import InfluxDBClient
        from influxdb_client.client.write_api import SYNCHRONOUS
        from influxdb_client.domain.write_precision import WritePrecision
    except ImportError as exc:  # pragma: no cover
        logger.error("influxdb-client not installed: %s", exc)
        return

    client = InfluxDBClient(url=cfg.url, token=cfg.token, org=cfg.org)
    write_api = client.write_api(write_options=SYNCHRONOUS)

    try:
        records = []
        for _, row in df_pandas.iterrows():
            ts_ns = int(row["window_end"].timestamp() * 1_000_000_000)
            records.append(
                f"vwap,"
                f"symbol={row['symbol']},"
                f"interval={row['interval']} "
                f"vwap={row['vwap']},"
                f"volume={row['total_volume']},"
                f"num_trades={int(row['num_trades'])}i "
                f"{ts_ns}"
            )
        write_api.write(
            bucket=cfg.bucket,
            org=cfg.org,
            record=records,
            write_precision=WritePrecision.NANOSECONDS,
        )
        logger.info("batch_id=%d: wrote %d VWAP points to InfluxDB", batch_id, len(records))
    except Exception as exc:
        logger.error("batch_id=%d: InfluxDB write failed: %s", batch_id, exc)
    finally:
        client.close()

def run(spark=None) -> None:
    """
    Khởi động Spark Structured Streaming VWAP job.

    Tất cả tham số (window size, watermark, trigger, Delta path, Kafka topic)
    đến từ ProcessingSettings — không hardcode ở đây.

    Args:
        spark: SparkSession tuỳ chọn — inject từ ngoài (dùng trong tests).
               None → tự tạo qua build_spark_session().
    """
    from pyspark.sql import functions as F
    from pyspark.sql.types import (
        DoubleType, LongType, StringType, StructField, StructType, TimestampType,
    )

    from dataops_processing.config.settings import get_settings
    from dataops_processing.config.spark_config import build_spark_session, get_kafka_options
    from dataops_processing.models.trade import TradeEvent

    cfg      = get_settings()
    vwap_cfg = cfg.vwap_job

    correlation_id = str(uuid.uuid4())
    logger.info("vwap_job.start correlation_id=%s", correlation_id)

    if spark is None:
        spark = build_spark_session("dataops-vwap-job")

    checkpoint_base = cfg.spark.checkpoint_dir("vwap_job")
    delta_path      = cfg.delta.silver_path(vwap_cfg.delta_table)

    logger.info(
        "VWAP job config — topic=%s window=%s/%s watermark=%s checkpoint=%s delta=%s",
        cfg.kafka.topic_trade_ticks,
        vwap_cfg.window_duration, vwap_cfg.slide_duration,
        vwap_cfg.watermark, checkpoint_base, delta_path,
    )

    # ── 1. Đọc từ Kafka ───────────────────────────────────────────────────────
    raw_stream = (
        spark.readStream
        .format("kafka")
        .options(**get_kafka_options())
        .load()
    )

    # ── 2. Deserialize — UDF gọi TradeEvent.from_kafka_bytes() ───────────────
    # Schema khớp TradeEvent.to_tuple() và CanonicalTrade trong .proto
    trade_schema = StructType([
        StructField("symbol", StringType(),  nullable=False),
        StructField("price",  DoubleType(),  nullable=False),
        StructField("size",   DoubleType(),  nullable=False),
        StructField("side",   StringType(),  nullable=True),
        StructField("ts_ms",  LongType(),    nullable=False),
    ])

    @F.udf(returnType=trade_schema)
    def parse_trade_udf(raw: bytes):
        event = TradeEvent.from_kafka_bytes(raw)
        if event is None:
            return None
        return event.to_tuple()

    parsed = (
        raw_stream
        .select(parse_trade_udf(F.col("value")).alias("trade"))
        .filter(F.col("trade").isNotNull())
        .select(
            F.col("trade.symbol").alias("symbol"),
            F.col("trade.price").alias("price"),
            F.col("trade.size").alias("size"),
            F.col("trade.side").alias("side"),
            (F.col("trade.ts_ms") / 1000).cast(TimestampType()).alias("event_time"),
            (F.col("trade.price") * F.col("trade.size")).alias("notional"),
        )
    )

    # ── 3. Watermark + Window aggregation ────────────────────────────────────
    with_watermark = parsed.withWatermark("event_time", vwap_cfg.watermark)

    vwap_agg = (
        with_watermark
        .groupBy(
            F.window("event_time", vwap_cfg.window_duration, vwap_cfg.slide_duration).alias("window"),
            F.col("symbol"),
        )
        .agg(
            (F.sum("notional") / F.sum("size")).alias("vwap"),
            F.sum("size").alias("total_volume"),
            F.count("*").alias("num_trades"),
        )
        .select(
            F.col("symbol"),
            F.col("window.start").alias("window_start"),
            F.col("window.end").alias("window_end"),
            F.lit(vwap_cfg.influx_interval_tag).alias("interval"),
            F.col("vwap"),
            F.col("total_volume"),
            F.col("num_trades"),
        )
    )

    # ── 4. Dual sink ──────────────────────────────────────────────────────────

    # Sink A: InfluxDB (foreachBatch — convert to pandas, gọi influxdb-client)
    influx_query = (
        vwap_agg
        .writeStream
        .outputMode(vwap_cfg.output_mode)
        .trigger(processingTime=vwap_cfg.trigger_interval)
        .foreachBatch(_write_to_influxdb)
        .option("checkpointLocation", f"{checkpoint_base}/influxdb")
        .queryName("vwap_to_influxdb")
        .start()
    )

    # Sink B: Delta Silver (append, partition by symbol)
    delta_query = (
        vwap_agg
        .writeStream
        .outputMode(vwap_cfg.output_mode)
        .trigger(processingTime=vwap_cfg.trigger_interval)
        .format("delta")
        .option("checkpointLocation", f"{checkpoint_base}/delta")
        .partitionBy("symbol")
        .start(delta_path)
    )

    logger.info("VWAP job streams started — awaiting termination correlation_id=%s", correlation_id)
    spark.streams.awaitAnyTermination()
    logger.info("VWAP job terminated correlation_id=%s", correlation_id)

# =============================================================================

def assign_volume_bars(rows: list[dict], threshold: float) -> list[dict]:
    """
    Assign bar index theo cumulative volume threshold.

    Args:
        rows:      List of candle dicts, sorted by ts_open_ms ascending.
                   Required keys: ts_open_ms, ts_close_ms, open, high, low, close, volume.
        threshold: Volume per bar (từ settings.volume_bars_job.thresholds[symbol]).

    Returns:
        List of VolumeBar-compatible dicts — 1 dict per bar với OHLCV aggregated.
        Empty list nếu rows rỗng.

    Algorithm:
        O(n) — duyệt tuần tự, cộng dồn volume.
        Khi cumsum >= threshold → đóng bar hiện tại, reset state cho bar tiếp theo.
        Partial bar cuối được emit dù chưa đủ threshold.
    """
    if not rows:
        return []

    bars: list[dict] = []
    bar_index = 0
    cum_vol   = 0.0
    bar_open  = rows[0]["open"]
    bar_high  = rows[0]["high"]
    bar_low   = rows[0]["low"]
    bar_open_ts  = rows[0]["ts_open_ms"]
    bar_close_ts = rows[0]["ts_close_ms"]
    num_candles  = 0

    for i, row in enumerate(rows):
        cum_vol      += row["volume"]
        bar_high      = max(bar_high, row["high"])
        bar_low       = min(bar_low,  row["low"])
        bar_close     = row["close"]
        bar_close_ts  = row["ts_close_ms"]
        num_candles  += 1

        if cum_vol >= threshold:
            bars.append({
                "bar_index":    bar_index,
                "open":         bar_open,
                "high":         bar_high,
                "low":          bar_low,
                "close":        bar_close,
                "total_volume": cum_vol,
                "num_candles":  num_candles,
                "bar_open_ts":  bar_open_ts,
                "bar_close_ts": bar_close_ts,
            })
            bar_index += 1
            # O(1) reset — dùng index thay vì rows.index(row) O(n)
            if i + 1 < len(rows):
                next_row     = rows[i + 1]
                bar_open     = next_row["open"]
                bar_high     = next_row["high"]
                bar_low      = next_row["low"]
                bar_open_ts  = next_row["ts_open_ms"]
                bar_close_ts = next_row["ts_close_ms"]
            cum_vol     = 0.0
            num_candles = 0

    # Partial bar cuối nếu còn data chưa đóng bar
    if cum_vol > 0 and num_candles > 0:
        bars.append({
            "bar_index":    bar_index,
            "open":         bar_open,
            "high":         bar_high,
            "low":          bar_low,
            "close":        bar_close,
            "total_volume": cum_vol,
            "num_candles":  num_candles,
            "bar_open_ts":  bar_open_ts,
            "bar_close_ts": bar_close_ts,
        })

    return bars

from __future__ import annotations

import logging.handlers

# ─── JsonFormatter ────────────────────────────────────────────────────────────

class JsonFormatter(logging.Formatter):
    """
    Serialize log records thành valid JSON cho log collectors (Loki / Grafana / ELK).

    Fields chuẩn: time, level, logger, message, module, line, process, thread.
    Extra fields được gắn qua logger.info("msg", extra={...}):
        correlation_id, job_name, symbol, component, operation, table

    correlation_id: UUID dùng để trace 1 job execution end-to-end.
    Cách dùng:
        logger.info("job started", extra={"correlation_id": run_id, "job_name": "vwap_job"})
    """

    def format(self, record: logging.LogRecord) -> str:
        payload: Dict[str, Any] = {
            "time":    self.formatTime(record, self.datefmt),
            "level":   record.levelname,
            "logger":  record.name,
            "message": record.getMessage(),
            "module":  record.module,
            "line":    record.lineno,
            "process": record.process,
            "thread":  record.threadName,
        }
        # Extra fields — phải khớp với các key dùng trong extra={} calls
        for field in ("correlation_id", "job_name", "symbol", "component",
                      "operation", "table", "source", "target"):
            value = getattr(record, field, None)
            if value is not None:
                payload[field] = value
        if record.exc_info:
            payload["exception"] = self.formatException(record.exc_info)
        return json.dumps(payload, ensure_ascii=False)


# ─── LoggerManager ────────────────────────────────────────────────────────────

class LoggerManager:
    """
    Singleton quản lý toàn bộ logging configuration cho dataops_processing.

    - Load dictConfig từ logger_config.yaml một lần duy nhất.
    - Patch filename handlers sang absolute path (tránh phụ thuộc working dir).
    - Cache logger instances để tránh tạo lại.
    - Fallback sang console-only khi YAML không tồn tại hoặc lỗi — pipeline không crash.
    """

    _instance: Optional["LoggerManager"] = None
    _loggers:  Dict[str, logging.Logger] = {}

    def __new__(cls) -> "LoggerManager":
        if cls._instance is None:
            cls._instance = super().__new__(cls)
            cls._instance._initialized = False
        return cls._instance

    def __init__(self) -> None:
        if getattr(self, "_initialized", False):
            return
        _here = Path(__file__).parent
        self._config_path = _here / "config" / "logger_config.yaml"
        self._log_storage = _here / "log_storage"

    # ── Public API ────────────────────────────────────────────────────────────

    def configure(self) -> None:
        """
        Load logger_config.yaml và áp dụng dictConfig.

        Gọi 1 lần tại entrypoint (runner.py). Gọi nhiều lần là no-op sau lần đầu.
        Fallback sang console-only nếu YAML không khả dụng — pipeline không crash.
        """
        if self._initialized:
            return

        try:
            self._log_storage.mkdir(parents=True, exist_ok=True)
        except OSError as exc:
            self._bootstrap_fallback()
            logging.getLogger(__name__).warning(
                "Cannot create log_storage dir %s: %s — console-only logging",
                self._log_storage, exc,
            )
            self._initialized = True
            return

        if not self._config_path.exists():
            self._bootstrap_fallback()
            logging.getLogger(__name__).warning(
                "logger_config.yaml not found at %s — console-only logging",
                self._config_path,
            )
            self._initialized = True
            return

        try:
            import yaml
            with open(self._config_path, "r", encoding="utf-8") as fh:
                cfg: Dict[str, Any] = yaml.safe_load(fh) or {}
            self._patch_file_handler_paths(cfg)
            logging.config.dictConfig(cfg)
            self._initialized = True
            logging.getLogger(__name__).debug(
                "Pipeline logging configured from %s", self._config_path,
            )
        except Exception as exc:
            self._bootstrap_fallback()
            logging.getLogger(__name__).warning(
                "Failed to load logger_config.yaml: %s — console-only logging", exc,
            )
            self._initialized = True

    def get_logger(self, name: str) -> logging.Logger:
        """
        Trả về logger theo name, có cache.

        Args:
            name: Thường là __name__ của module gọi vào.
                  Ví dụ: 'dataops_processing.jobs.vwap_job'

        Returns:
            logging.Logger đã được route theo hierarchy trong logger_config.yaml.
        """
        if not self._initialized:
            self.configure()
        if name not in self._loggers:
            self._loggers[name] = logging.getLogger(name)
        return self._loggers[name]

    # ── Internal helpers ──────────────────────────────────────────────────────

    def _bootstrap_fallback(self) -> None:
        """Console-only logging tối thiểu — đảm bảo pipeline không crash."""
        if logging.root.handlers:
            return
        logging.basicConfig(
            level=logging.INFO,
            format="%(asctime)s [%(levelname)s] %(name)s - %(message)s",
            datefmt="%Y-%m-%dT%H:%M:%S",
            stream=sys.stdout,
        )

    def _patch_file_handler_paths(self, cfg: Dict[str, Any]) -> None:
        """
        Thay relative filename trong RotatingFileHandler thành absolute path.

        Tránh phụ thuộc working directory khi chạy từ các vị trí khác nhau.
        """
        for handler_cfg in cfg.get("handlers", {}).values():
            if not isinstance(handler_cfg, dict):
                continue
            if "filename" in handler_cfg:
                original = Path(handler_cfg["filename"])
                if not original.is_absolute():
                    handler_cfg["filename"] = str(self._log_storage / original.name)

    được route đúng handler theo hierarchy trong logger_config.yaml.

## Tóm tắt

### Cấu trúc log mới

```
dataops_processing/
└── log/
    ├── __init__.py
    ├── logger_setup.py          ← LoggerManager singleton + JsonFormatter
    ├── config/
    │   └── logger_config.yaml  ← dictConfig — route theo __name__ prefix
    └── log_storage/
        └── .gitkeep            ← RotatingFileHandler ghi vào đây
```

### Pattern chuẩn — nhất quán với `dataops_batch_api`

| Khía cạnh | Trước (sai) | Sau (đúng) |
|---|---|---|
| Import logger | `logging.getLogger(__name__)` | `logger_manager.get_logger(__name__)` |
| Setup tại entrypoint | `logging.basicConfig(...)` trong runner | `setup_pipeline_logging()` → load YAML |
| Format | Plain text `%(message)s` | `JsonFormatter` → JSON cho Loki/Grafana |
| Log trong except | `logger.error("... %s", exc)` | `logger.error("... %s", exc, exc_info=True)` — kèm stacktrace |
| Routing | Không có | `logger_config.yaml` — `dataops_processing.jobs.*` → `jobs.log`, `.config.*` → `config.log`, `.models.*` → `models.log` |
| Extra fields | Không có | `extra={"correlation_id": ..., "job_name": ..., "symbol": ...}` |
| Fallback | crash nếu không setup | `_bootstrap_fallback()` — console-only, pipeline không crash |

### Extra fields dùng trong log calls

```python
# Trong run() của mỗi job — gắn correlation_id để trace end-to-end
logger.info("vwap_job.start correlation_id=%s", correlation_id)

# Với structured extra (Loki sẽ index được):
logger.info("batch done", extra={"correlation_id": run_id, "job_name": "vwap_job", "symbol": "BTCUSDT"})

# Trong except — luôn có exc_info=True để stacktrace vào JSON:
logger.error("symbol=%s failed — %s", sym, exc, exc_info=True)
```
